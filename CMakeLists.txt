cmake_minimum_required(VERSION 3.24)
project(tinyinfer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(OpenMP REQUIRED)
option(TINYC_CUDA "Build CUDA backend" OFF)
if(TINYC_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_ARCHITECTURES 86)
endif()
# Generated models use this same CMake entry point and the already-built shared backend.
if(TINYC_JIT_SOURCE)
  add_library(generated SHARED "${TINYC_JIT_SOURCE}")
  target_include_directories(generated PRIVATE tinyc/runtime backend/include)
  target_link_libraries(generated PRIVATE OpenMP::OpenMP_CXX "${TINYC_BACKEND_LIBRARY}")
  target_compile_options(generated PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-O3;-march=native;-fno-math-errno>" "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--use_fast_math;-allow-unsupported-compiler>")
  if(TINYC_CUDA)
    target_link_libraries(generated PRIVATE CUDA::cudart)
  endif()
  get_filename_component(backend_dir "${TINYC_BACKEND_LIBRARY}" DIRECTORY)
  set_target_properties(generated PROPERTIES PREFIX "" OUTPUT_NAME "${TINYC_JIT_NAME}" LIBRARY_OUTPUT_DIRECTORY "${TINYC_JIT_OUTPUT}" BUILD_RPATH "${backend_dir}")
  return()
endif()
add_library(tinyinfer_backend SHARED backend/runtime.cpp kernels/src/cpu/matmul.cpp kernels/src/cpu/softmax.cpp kernels/src/cpu/layernorm.cpp)
target_include_directories(tinyinfer_backend PUBLIC kernels/include backend/include)
target_link_libraries(tinyinfer_backend PUBLIC OpenMP::OpenMP_CXX)
target_compile_options(tinyinfer_backend PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:-O3;-march=native;-ffp-contract=fast;-fno-math-errno>")
if(TINYC_CUDA)
  target_sources(tinyinfer_backend PRIVATE kernels/src/cuda/matmul.cu kernels/src/cuda/softmax.cu kernels/src/cuda/layernorm.cu kernels/src/cuda/bench_api.cu)
  target_compile_definitions(tinyinfer_backend PUBLIC IK_HAVE_CUDA=1)
  target_compile_options(tinyinfer_backend PRIVATE "$<$<COMPILE_LANGUAGE:CUDA>:-O3;--use_fast_math;-allow-unsupported-compiler>")
  target_link_libraries(tinyinfer_backend PUBLIC CUDA::cudart)
endif()
find_package(Python3 COMPONENTS Interpreter Development.Module REQUIRED)
execute_process(COMMAND ${Python3_EXECUTABLE} -m pybind11 --cmakedir OUTPUT_VARIABLE pybind_dir OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
find_package(pybind11 CONFIG REQUIRED HINTS "${pybind_dir}")
pybind11_add_module(ikernels kernels/python/bindings.cpp)
target_link_libraries(ikernels PRIVATE tinyinfer_backend)
enable_testing()
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz URL_HASH SHA256=7b42b4d6ed48810c5362c265a17faebe90dc2373c885e5216439d37927f02926 DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
set(BUILD_GMOCK OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
include(GoogleTest)
add_executable(test_cpu kernels/tests/test_cpu.cpp)
target_link_libraries(test_cpu PRIVATE tinyinfer_backend GTest::gtest_main)
gtest_discover_tests(test_cpu)
if(TINYC_CUDA)
  add_executable(test_cuda kernels/tests/test_cuda.cpp)
  target_link_libraries(test_cuda PRIVATE tinyinfer_backend GTest::gtest_main)
  gtest_discover_tests(test_cuda DISCOVERY_MODE PRE_TEST)
endif()
