cmake_minimum_required(VERSION 3.24)
project(tinyinfer_server LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(Threads REQUIRED)
option(TI_SERVER_NATIVE "Link the same standalone native GPT-2 library" OFF)
option(TI_SERVER_TSAN "Run CPU queue/protocol/server with ThreadSanitizer" OFF)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The server requires Linux epoll/eventfd (WSL2 supported)")
endif()
add_library(ti_server_options INTERFACE)
target_include_directories(ti_server_options INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(ti_server_options INTERFACE -Wall -Wextra -Wpedantic)
target_link_libraries(ti_server_options INTERFACE Threads::Threads)
if(TI_SERVER_TSAN)
  target_compile_options(ti_server_options INTERFACE -fsanitize=thread -fno-omit-frame-pointer -g -O1)
  target_link_options(ti_server_options INTERFACE -fsanitize=thread)
endif()
add_executable(tinyinfer_server src/main.cpp)
target_link_libraries(tinyinfer_server PRIVATE ti_server_options)
if(TI_SERVER_NATIVE)
  if(NOT TARGET tinyinfer_llm)
    message(FATAL_ERROR "Enable from root after adding native tinyinfer_llm; no stub fallback")
  endif()
  target_compile_definitions(tinyinfer_server PRIVATE TI_SERVER_NATIVE=1)
  target_include_directories(tinyinfer_server PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../llm/include)
  target_link_libraries(tinyinfer_server PRIVATE tinyinfer_llm)
endif()
if(BUILD_TESTING)
  add_executable(ti_server_unit tests/unit.cpp)
  target_compile_options(ti_server_unit PRIVATE -UNDEBUG)
  target_link_libraries(ti_server_unit PRIVATE ti_server_options)
  add_test(NAME v2_server_queue_protocol COMMAND ti_server_unit)
  set_tests_properties(v2_server_queue_protocol PROPERTIES TIMEOUT 60)
  add_executable(ti_server_transport tests/transport.cpp)
  target_compile_options(ti_server_transport PRIVATE -UNDEBUG)
  target_link_libraries(ti_server_transport PRIVATE ti_server_options)
  add_test(NAME v2_server_transport_regressions COMMAND ti_server_transport)
  set_tests_properties(v2_server_transport_regressions PROPERTIES TIMEOUT 10)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME v2_server_e2e COMMAND ${Python3_EXECUTABLE} -m unittest server.tests.test_e2e)
  set_tests_properties(v2_server_e2e PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/..
    ENVIRONMENT "TI_SERVER_BINARY=$<TARGET_FILE:tinyinfer_server>;PYTHONDONTWRITEBYTECODE=1" TIMEOUT 60)
endif()
